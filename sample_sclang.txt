157	// We accept a / b / l prefixes (the static groups) and reject
158	// anything else, so a stray `$` inside an expression never invents
159	// a lane.
160	fn orbits_in_source(source: &str) -> Vec<String> {
161	    let mut out = Vec::new();
162	    for line in source.lines() {
163	        let trimmed = line.trim_start();
164	        // Must start with a letter we recognize, then digits, then `$`.
165	        let Some(rest) = trimmed.strip_prefix(|c: char| matches!(c, 'a' | 'b' | 'l')) else {
166	            continue;
167	        };
168	        let digits_end = rest
169	            .find(|c: char| !c.is_ascii_digit())
170	            .unwrap_or(rest.len());
171	        if digits_end == 0 {
172	            continue;
173	        }
174	        let after = &rest[digits_end..];
175	        // Tolerate spaces between the orbit name and `$`.
176	        let after_trim = after.trim_start();
177	        if after_trim.starts_with('$') {
178	            let orbit = format!("{}{}", &trimmed[..1], &rest[..digits_end]);
179	            if !out.contains(&orbit) {
180	                out.push(orbit);
181	            }
182	        }
183	    }
184	    out
185	}
